TradingView Pine Script v5: 7-MA Rainbow Ribbon with Signals 
//@version=5
indicator("Vault Rainbow MA Ribbon with Signals", shorttitle="Rainbow_Ribbon_Signals", overlay=true)

src     = input.source(close, title="Source")
ma_type = input.string("EMA", title="MA Type", options=["EMA", "SMA", "HMA", "RMA", "WMA", "VWMA"])

len1 = input.int(8, "MA 1 (Fast)"), len2 = input.int(13, "MA 2"), len3 = input.int(21, "MA 3")
len4 = input.int(34, "MA 4"),        len5 = input.int(55, "MA 5"), len6 = input.int(89, "MA 6")
len7 = input.int(233, "MA 7 (Macro)")

calc_ma(t, s, l) =>
    t == "EMA" ? ta.ema(s, l) : t == "HMA" ? ta.hma(s, l) : t == "VWMA" ? ta.vwma(s, l) : ta.sma(s, l)

ma1 = calc_ma(ma_type, src, len1), ma2 = calc_ma(ma_type, src, len2), ma3 = calc_ma(ma_type, src, len3)
ma4 = calc_ma(ma_type, src, len4), ma5 = calc_ma(ma_type, src, len5), ma6 = calc_ma(ma_type, src, len6)
ma7 = calc_ma(ma_type, src, len7)

p1 = plot(ma1, color=#ff0055, linewidth=2), p3 = plot(ma3, color=#ffcc00, linewidth=1)
p4 = plot(ma4, color=#33cc33, linewidth=1), p5 = plot(ma5, color=#00cccc, linewidth=2)
plot(ma6, color=#0066ff, linewidth=2), plot(ma7, color=#990099, linewidth=3)

fill(p1, p3, color=ma1 > ma3 ? color.new(color.green, 85) : color.new(color.red, 85))
fill(p4, p5, color=ma4 > ma5 ? color.new(color.teal, 88) : color.new(color.maroon, 88))

bull_cross = ta.crossover(ma1, ma2)
bear_cross = ta.crossunder(ma1, ma2)

plotshape(bull_cross, style=shape.circle, location=location.belowbar, color=color.green, size=size.small)
plotshape(bear_cross, style=shape.circle, location=location.abovebar, color=color.red, size=size.small)

alertcondition(bull_cross, "Bullish Rainbow Cross", "MA1 crossed above MA2")
alertcondition(bear_cross, "Bearish Rainbow Cross", "MA1 crossed below MA2")
